def fig_extent_hit_miss(out_dir, skips):

    panels = [("LF vs HF", b["lf"]), ("LSG-Max vs HF", b["pred"])]
    for ax, (title, field):
        cat = _extent_category(b["hf"], field, DEPTH_TAU_M)   # 0 dry,1 hit,2 miss,3 FA at tau=0.03
        _scatter_categories(ax, b["xy"], cat, s=0.55 or 0.25 if n>400k)
    fig.suptitle(f"{case} · event {eid} · inundation-extent classification (τ=0.03 m)")

def _extent_category(hf, pred, tau):
    hf_w = hf >= tau; pr_w = pred >= tau
    cat[hf_w & pr_w]=1; cat[hf_w & ~pr_w]=2; cat[~hf_w & pr_w]=3
